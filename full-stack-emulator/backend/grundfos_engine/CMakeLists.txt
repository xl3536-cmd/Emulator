cmake_minimum_required(VERSION 3.16)
project(grundfos_pump_emulator LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

# The Pi build is an MS/TP device. BIP is only for developer protocol tests.
option(PUMP_TEST_BIP "Build a BACnet/IP test executable instead of MS/TP" OFF)
set(BACNET_STACK_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(BAC_ROUTING OFF CACHE BOOL "" FORCE)
set(BACNET_BACKUP_RESTORE OFF CACHE BOOL "" FORCE)
set(BACNET_SEGMENTATION_ENABLED OFF CACHE BOOL "" FORCE)
set(BACDL_BIP ${PUMP_TEST_BIP} CACHE BOOL "" FORCE)
if(PUMP_TEST_BIP)
    set(BACDL_MSTP OFF CACHE BOOL "" FORCE)
else()
    set(BACDL_MSTP ON CACHE BOOL "" FORCE)
endif()
foreach(link BIP6 ZIGBEE ETHERNET ARCNET BSC)
    set(BACDL_${link} OFF CACHE BOOL "" FORCE)
endforeach()
set(BACNET_DEFINES "MAX_APDU=480;BACNET_SEGMENTATION_ENABLED=0" CACHE STRING "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/bacnet-stack/CMakeLists.txt")
    message(FATAL_ERROR "Missing vendor/bacnet-stack. Run python3 build.py first.")
endif()
add_subdirectory(vendor/bacnet-stack EXCLUDE_FROM_ALL)
target_compile_definitions(bacnet-stack PUBLIC MAX_APDU=480 BACNET_SEGMENTATION_ENABLED=0)

add_library(pump_objects STATIC pump_objects.c)
target_link_libraries(pump_objects PUBLIC bacnet-stack::bacnet-stack)
target_include_directories(pump_objects PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(pump_objects PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(pump_emulator main.c)
target_link_libraries(pump_emulator PRIVATE pump_objects)
if(PUMP_TEST_BIP)
    target_compile_definitions(pump_emulator PRIVATE PUMP_TEST_BIP=1)
endif()
enable_testing()
add_executable(test_pump_objects tests/test_pump_objects.c)
target_link_libraries(test_pump_objects PRIVATE pump_objects)
add_test(NAME pump_objects COMMAND test_pump_objects)

if(NOT PUMP_TEST_BIP AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(pump_worker main.c pump_bus_link.c)
    target_compile_definitions(pump_worker PRIVATE PUMP_GROUP_WORKER=1)
    target_link_libraries(pump_worker PRIVATE pump_objects)
    foreach(symbol dlmstp_init dlmstp_cleanup dlmstp_send_pdu dlmstp_receive)
        target_link_options(pump_worker PRIVATE "LINKER:--wrap=${symbol}")
    endforeach()
    add_executable(mstp_bus mstp_bus.c)
    target_include_directories(mstp_bus PRIVATE vendor/bacnet-stack/ports/linux)
    target_link_libraries(mstp_bus PRIVATE bacnet-stack::bacnet-stack)
    add_executable(test_mstp_bus tests/test_mstp_bus.c)
    target_include_directories(test_mstp_bus PRIVATE vendor/bacnet-stack/ports/linux)
    target_link_libraries(test_mstp_bus PRIVATE bacnet-stack::bacnet-stack)
    add_test(NAME mstp_group_bus COMMAND test_mstp_bus)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(mstp_bus PRIVATE -Wall -Wextra)
        target_compile_options(pump_worker PRIVATE -Wall -Wextra)
    endif()
endif()

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME group_configuration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_group_config.py)
endif()
